cmake_minimum_required(VERSION 3.20)
project(meta_nano_extract LANGUAGES CXX)
find_package(OpenVDB REQUIRED)
find_path(NANOVDB_INCLUDE_DIR nanovdb/tools/CreateNanoGrid.h REQUIRED)
add_executable(meta_nano_extract nano_extract.cpp)
target_compile_features(meta_nano_extract PRIVATE cxx_std_17)
target_compile_definitions(meta_nano_extract PRIVATE NANOVDB_USE_OPENVDB)
target_include_directories(meta_nano_extract PRIVATE ${NANOVDB_INCLUDE_DIR})
target_link_libraries(meta_nano_extract PRIVATE OpenVDB::openvdb)
